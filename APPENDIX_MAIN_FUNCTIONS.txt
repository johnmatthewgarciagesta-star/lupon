ALGORITHM #3: FUNCTION 3: SEQUENTIAL CASE FOLDER AUTO-NUMBERING ALGORITHM
File: app/Http/Controllers/DocumentController.php
Signature: DocumentController::getNextAvailableCaseFolderNumber()
Description: Calculates the next available zero-padded sequential folder index (case-XXX) across cases and documents tables, checking soft-deleted entries to avoid identifier collisions.
--------------------------------------------------------------------------------
  1 | public function getNextAvailableCaseFolderNumber(): string
  2 | {
  3 |     try {
  4 |         $caseFolders = \App\Models\LuponCase::withTrashed()
  5 |             ->whereNotNull('folder_name')
  6 |             ->where('folder_name', '!=', '')
  7 |             ->pluck('folder_name')
  8 |             ->toArray();
  9 | 
 10 |         $docFolders = \App\Models\Document::withTrashed()
 11 |             ->whereNotNull('folder_name')
 12 |             ->where('folder_name', '!=', '')
 13 |             ->pluck('folder_name')
 14 |             ->toArray();
 15 | 
 16 |         $allFolders = array_unique(array_filter(array_merge($caseFolders, $docFolders)));
 17 | 
 18 |         $maxIndex = 0;
 19 |         foreach ($allFolders as $folder) {
 20 |             $folder = trim((string)$folder);
 21 |             if (preg_match('/(?:case|folder)[-_ ]*0*(\d+)/i', $folder, $matches)) {
 22 |                 $num = (int)$matches[1];
 23 |                 if ($num > $maxIndex) $maxIndex = $num;
 24 |             } elseif (preg_match('/^0*(\d+)$/', $folder, $matches)) {
 25 |                 $num = (int)$matches[1];
 26 |                 if ($num > $maxIndex) $maxIndex = $num;
 27 |             }
 28 |         }
 29 | 
 30 |         $nextIndex = max($maxIndex + 1, 1);
 31 |         $candidateFolder = 'case-' . str_pad($nextIndex, 3, '0', STR_PAD_LEFT);
 32 | 
 33 |         while (
 34 |             \App\Models\LuponCase::withTrashed()->whereRaw('LOWER(folder_name) = ?', [strtolower($candidateFolder)])->exists() ||
 35 |             \App\Models\Document::withTrashed()->whereRaw('LOWER(folder_name) = ?', [strtolower($candidateFolder)])->exists()
 36 |         ) {
 37 |             $nextIndex++;
 38 |             $candidateFolder = 'case-' . str_pad($nextIndex, 3, '0', STR_PAD_LEFT);
 39 |         }
 40 | 
 41 |         return $candidateFolder;
 42 |     } catch (\Throwable $e) {
 43 |         \Log::error('Error calculating folder number: ' . $e->getMessage());
 44 |         $fallback = (\App\Models\LuponCase::withTrashed()->count() ?: 0) + 1;
 45 |         return 'case-' . str_pad($fallback, 3, '0', STR_PAD_LEFT);
 46 |     }
 47 | }
